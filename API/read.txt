-- 1) Header: fee + student + academic info
SELECT f.id, f.date, f.year, f.term, f.amount, f.discount, f.total, f.paid_amount,
       (f.total - f.paid_amount) AS balance,
       u.name  AS student_name, u.code AS student_code,
       c.name  AS class_name, m.name AS major, g.name AS generation,
       p.name  AS programme, s.name AS scholarship, fs.name AS schedule_name
FROM fees f
JOIN users u          ON u.id  = f.user_id
JOIN classes c        ON c.id  = f.class_id
JOIN majors m         ON m.id  = f.major_id
JOIN generations g    ON g.id  = f.generation_id
JOIN programmes p     ON p.id  = f.programme_id
JOIN fee_schedules fs ON fs.id = f.fee_schedule_id
LEFT JOIN scholarships s ON s.id = f.scholarship_id
WHERE f.id = :fee_id AND f.deleted_at IS NULL;

-- 2) Installments
SELECT sequence_no, due_date, amount, status
FROM installments WHERE fee_id = :fee_id ORDER BY sequence_no;

-- 3) Payments received
SELECT code, date, method, reference, total
FROM fee_transactions
WHERE fee_id = :fee_id AND active = 1 ORDER BY date;
